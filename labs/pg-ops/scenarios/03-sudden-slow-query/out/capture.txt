# orders 100만 행: 대형 고객 1이 40%, 중형 고객 2~11이 3%씩, 소형 고객 2만 명이 나머지

postgres=# CREATE TABLE orders (id bigint PRIMARY KEY, customer_id int, status text, amount int);
CREATE TABLE

postgres=# INSERT INTO orders
postgres-# SELECT g,
postgres-#        CASE WHEN g % 10 < 4 THEN 1
postgres-#             WHEN g % 10 < 7 THEN 2 + (g / 10) % 10
postgres-#             ELSE 12 + (g / 10) % 20000 END,
postgres-#        'done', g % 997
postgres-# FROM generate_series(1, 1000000) g;
INSERT 0 1000000

postgres=# CREATE TABLE shipments (id bigserial PRIMARY KEY, order_id bigint, state text);
CREATE TABLE

postgres=# INSERT INTO shipments (order_id, state) SELECT g, 'delivered' FROM generate_series(1, 1000000) g;
INSERT 0 1000000

postgres=# CREATE INDEX ON orders (status);
CREATE INDEX

postgres=# CREATE INDEX ON orders (customer_id);
CREATE INDEX

postgres=# CREATE INDEX ON shipments (state);
CREATE INDEX
