{{- /* 터미널 세션 블록: "$ "와 이어지는 줄 "> " 뒤만 bash로 하이라이팅하고, 나머지 줄은 결과(go)로 표시한다. Chroma의 console 렉서는 "#"으로 시작하는 결과 줄(설정 파일의 주석 등)을 root 프롬프트로 읽어서 직접 나눈다. */ -}} {{- $out := slice -}} {{- range split (strings.TrimRight "\n" .Inner) "\n" -}} {{- $m := findRESubmatch `^([$>]) (.*)$` . 1 -}} {{- if $m -}} {{- $m = index $m 0 -}} {{- $cmd := transform.Highlight (index $m 2) "bash" "hl_inline=true" | replaceRE `^]*>|$` "" -}} {{- $out = $out | append (printf `%s %s ` (index $m 1) $cmd) -}} {{- else -}} {{- $out = $out | append (printf `%s ` (htmlEscape .)) -}} {{- end -}} {{- end -}} 
{{ delimit $out "" | safeHTML }}
